from fastapi import APIRouter, Request, Depends, Form, HTTPException from fastapi.responses import HTMLResponse from sqlalchemy.ext.asyncio import AsyncSession from datetime import datetime from web_admin.dependencies import get_async_session, templates from models.item import Item router = APIRouter(prefix="/admin/assortment", tags=["sales"]) @router.get("/sale/{item_id}", response_class=HTMLResponse) async def get_sale_form( item_id: int, request: Request, session: AsyncSession = Depends(get_async_session) ): """Открытие модалки продажи""" item = await session.get(Item, item_id) if not item or item.is_sold: raise HTTPException(status_code=404, detail="Товар не найден или уже продан") return templates.TemplateResponse( "assortment_sale_item.html", { "request": request, "item": item, "error": None, "form_data": None, } ) @router.post("/sale/{item_id}", response_class=HTMLResponse) async def process_sale( item_id: int, request: Request, session: AsyncSession = Depends(get_async_session), sale_price: float = Form(...), sale_payment_type: str = Form(...), sale_payment_amount: float = Form(0), sale_change: float = Form(0), sale_change_type: str = Form(""), sale_platform: str = Form(""), sale_full_name: str = Form(""), sale_phone: str = Form(""), sale_birth_date: str = Form(""), sale_comment: str = Form(""), ): """Обработка продажи товара""" item = await session.get(Item, item_id) if not item or item.is_sold: return templates.TemplateResponse( "assortment_sale_item.html", { "request": request, "item": item, "error": "Товар не найден или уже продан", "form_data": { "sale_price": sale_price, "sale_payment_type": sale_payment_type, "sale_payment_amount": sale_payment_amount, "sale_change": sale_change, "sale_change_type": sale_change_type, "sale_platform": sale_platform, "sale_full_name": sale_full_name, "sale_phone": sale_phone, "sale_birth_date": sale_birth_date, "sale_comment": sale_comment, }, }, status_code=400, ) try: # === Записываем данные продажи в Item === item.is_sold = True item.sale_price = sale_price item.sale_payment_type = sale_payment_type item.sale_payment_amount = sale_payment_amount or sale_price item.sale_change = sale_change item.sale_change_type = sale_change_type item.sale_platform = sale_platform item.sale_full_name = sale_full_name item.sale_phone = sale_phone item.sale_birth_date = sale_birth_date await session.commit() # Успешная продажа — модалка закроется автоматически return HTMLResponse(status_code=200) except Exception as e: await session.rollback() return templates.TemplateResponse( "assortment_sale_item.html", { "request": request, "item": item, "error": f"Ошибка при продаже: {str(e)}", "form_data": { "sale_price": sale_price, "sale_payment_type": sale_payment_type, "sale_payment_amount": sale_payment_amount, "sale_change": sale_change, "sale_change_type": sale_change_type, "sale_platform": sale_platform, "sale_full_name": sale_full_name, "sale_phone": sale_phone, "sale_birth_date": sale_birth_date, "sale_comment": sale_comment, }, }, status_code=400, )
